# Load libraries
library(rpart)

Model Data

Select predictors and total charge.

# Create new dataset with only the variables I need
df2 = read_csv("Data/midterm.csv") %>% select(tot, age, sex, raceethn, provider, moa, mod, admtype, campus, los)

Remove Missing

# Filter to variables that are not missing
df2 = df2 %>% filter(!is.na(raceethn), !is.na(admtype)) %>% drop_na()

Q1: Target and Split

Target = high/low charge (median split); 10% train.

# Make the target into high and low splitting on median
df2 = df2 %>% mutate(target = ifelse(tot < median(tot), "low", "high")) %>% select(-tot)

# Mutate the variables to be factors
df2 = df2 %>% mutate(target = factor(target),
                     sex = factor(sex), raceethn = factor(raceethn),
                     provider = factor(provider), moa = factor(moa),
                     mod = factor(mod), admtype = factor(admtype),
                     campus = factor(campus))

# Set seed to be replicable and the data partition for training and testing
set.seed(2018)
splitIndex = createDataPartition(df2$target, p = 0.1, list = FALSE)
df_train = df2[splitIndex, ]
df_test = df2[-splitIndex, ]

Challenges

Challenges and Fixes

  • Large SAS file was slow to load, filtered to 2018, kept needed columns, saved a smaller CSV.
  • Missing values existed so I dropped columns with NAs; used drop_na() for modeling data.
  • Provider codes were just numbers so I mapped them to hospital names with case_when().
  • Random forest took a long time to train, so I trained on a 10% sample with tuneLength = 3.
  • Warnings/messages cluttered the output so I had to include message=FALSE, warning=FALSE on each chunk.
